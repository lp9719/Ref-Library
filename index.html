  <script>

    // --- FIREBASE INITIALIZATION ---

    const firebaseConfig = typeof __firebase_config !== 'undefined'

      ? JSON.parse(__firebase_config)

      : {

          // Configuração padrão de Fallback (Mock) caso esteja rodando offline/local sem backend configurado

          apiKey: "demo", authDomain: "demo.firebaseapp.com", projectId: "demo"

        };

    const app = firebase.initializeApp(firebaseConfig);

    const auth = firebase.auth();

    const db = firebase.firestore();

    const appId = typeof __app_id !== 'undefined' ? __app_id : 'refhub-app';

    // --- APPLICATION STATE ---

    let currentFolderId = 'root';

    let folderPath = [{ id: 'root', name: 'Início' }];

    let folders = [];

    let references = [];

    let currentUser = null;

    let uploadType = 'url';

    let activeRefId = null;

    // --- DOM ELEMENTS ---

    const breadcrumbEl = document.getElementById('breadcrumb');

    const foldersContainer = document.getElementById('foldersContainer');

    const referencesGrid = document.getElementById('referencesGrid');

    const emptyState = document.getElementById('emptyState');

    const searchInput = document.getElementById('searchInput');

    const clearSearch = document.getElementById('clearSearch');

    const itemCount = document.getElementById('itemCount');

    const syncStatus = document.getElementById('syncStatus');

    // Modals

    const modalFolder = document.getElementById('modalFolder');

    const modalRef = document.getElementById('modalRef');

    const modalLightbox = document.getElementById('modalLightbox');

    // Forms

    const formFolder = document.getElementById('formFolder');

    const formRef = document.getElementById('formRef');

    // --- AUTHENTICATION & INITIALIZATION ---

    async function initApp() {

      try {

        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

          await auth.signInWithCustomToken(__initial_auth_token);

        } else {

          await auth.signInAnonymously();

        }

      } catch (e) {

        console.warn("Auth fallback to local mode:", e);

      }

      auth.onAuthStateChanged((user) => {

        currentUser = user;

        updateSyncBadge(true);

        loadData();

      });

     

      lucide.createIcons();

    }

    function updateSyncBadge(isSynced) {

      if(isSynced) {

        syncStatus.innerHTML = `<i data-lucide="cloud-check" class="w-3.5 h-3.5"></i><span>Sincronizado</span>`;

        syncStatus.className = "flex items-center gap-1.5 text-xs text-emerald-400 bg-emerald-950/60 border border-emerald-800/50 px-2.5 py-1 rounded-full";

      } else {

        syncStatus.innerHTML = `<i data-lucide="refresh-cw" class="w-3.5 h-3.5 animate-spin"></i><span>Salvando...</span>`;

        syncStatus.className = "flex items-center gap-1.5 text-xs text-amber-400 bg-amber-950/60 border border-amber-800/50 px-2.5 py-1 rounded-full";

      }

      lucide.createIcons();

    }

    // --- DATA SNAPSHOT LISTENERS ---

    function loadData() {

      if (!currentUser) return;

      const foldersCol = firebase.firestore().collection(db, 'artifacts', appId, 'users', currentUser.uid, 'folders');

      const refsCol = firebase.firestore().collection(db, 'artifacts', appId, 'users', currentUser.uid, 'references');

      // Sync Folders

      foldersCol.onSnapshot((snapshot) => {

        folders = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));

        render();

      }, (err) => console.error("Folders sync error:", err));

      // Sync References

      refsCol.onSnapshot((snapshot) => {

        references = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));

        render();

      }, (err) => console.error("Refs sync error:", err));

    }

    // --- RENDER LOGIC ---

    function render() {

      const query = searchInput.value.toLowerCase().trim();

      renderBreadcrumb();

      if (query !== '') {

        renderSearchResults(query);

        return;

      }

      // Filter for Current Level

      const currentFolders = folders.filter(f => f.parentId === currentFolderId);

      const currentRefs = references.filter(r => r.folderId === currentFolderId);

      // Render Subfolders

      foldersContainer.innerHTML = '';

      currentFolders.forEach(folder => {

        const fEl = document.createElement('div');

        fEl.className = 'group flex items-center justify-between p-3 bg-gray-900 border border-gray-800 rounded-xl hover:border-indigo-600/50 hover:bg-gray-850 cursor-pointer transition-all';

        fEl.innerHTML = `

          <div class="flex items-center gap-2.5 min-w-0" onclick="navigateToFolder('${folder.id}', '${escapeQuotes(folder.name)}')">

            <i data-lucide="folder" class="w-5 h-5 text-indigo-400 shrink-0 group-hover:fill-indigo-400/20"></i>

            <span class="text-sm font-medium text-gray-200 truncate">${folder.name}</span>

          </div>

          <button onclick="deleteFolder(event, '${folder.id}')" class="opacity-0 group-hover:opacity-100 p-1 text-gray-500 hover:text-red-400 transition-opacity">

            <i data-lucide="trash-2" class="w-4 h-4"></i>

          </button>

        `;

        foldersContainer.appendChild(fEl);

      });

      // Render References Masonry Grid

      referencesGrid.innerHTML = '';

      currentRefs.forEach(ref => {

        const rEl = createReferenceCard(ref);

        referencesGrid.appendChild(rEl);

      });

      // Update Counts & States

      const totalItems = currentFolders.length + currentRefs.length;

      itemCount.innerText = `${totalItems} item(ns)`;

      if (totalItems === 0) {

        emptyState.classList.remove('hidden');

      } else {

        emptyState.classList.add('hidden');

      }

      lucide.createIcons();

    }

    function createReferenceCard(ref) {

      const card = document.createElement('div');

      card.className = 'break-inside-avoid mb-4 group relative bg-gray-900 border border-gray-800 rounded-xl overflow-hidden hover:border-gray-700 transition-all shadow-md';

     

      const tagsHtml = (ref.tags || []).map(t => `<span class="bg-indigo-950/80 border border-indigo-800/40 text-indigo-300 text-[10px] px-1.5 py-0.5 rounded">#${t}</span>`).join(' ');

      card.innerHTML = `

        <div class="relative overflow-hidden cursor-pointer bg-gray-950" onclick="openLightbox('${ref.id}')">

          <img src="${ref.url}" alt="${ref.title}" loading="lazy" class="w-full object-cover group-hover:scale-105 transition-transform duration-300 max-h-96" onerror="this.src='https://placehold.co/400x300/1e293b/94a3b8?text=Imagem+Indispon%C3%ADvel'">

          <div class="absolute inset-0 bg-gradient-to-t from-black/80 via-transparent to-transparent opacity-0 group-hover:opacity-100 transition-opacity flex items-end p-3">

            <p class="text-xs text-white font-medium truncate">${ref.title}</p>

          </div>

        </div>

        <div class="p-2.5 flex flex-col gap-1.5">

          <h3 class="text-xs font-semibold text-gray-200 truncate">${ref.title}</h3>

          ${tagsHtml ? `<div class="flex flex-wrap gap-1">${tagsHtml}</div>` : ''}

        </div>

      `;

      return card;

    }

    function renderSearchResults(query) {

      foldersContainer.innerHTML = '';

      referencesGrid.innerHTML = '';

      const matchedRefs = references.filter(r =>

        r.title.toLowerCase().includes(query) ||

        (r.tags && r.tags.some(t => t.toLowerCase().includes(query)))

      );

      matchedRefs.forEach(ref => {

        const card = createReferenceCard(ref);

        referencesGrid.appendChild(card);

      });

      itemCount.innerText = `${matchedRefs.length} resultado(s)`;

      if (matchedRefs.length === 0) {

        emptyState.classList.remove('hidden');

      } else {

        emptyState.classList.add('hidden');

      }

      lucide.createIcons();

    }

    function renderBreadcrumb() {

      breadcrumbEl.innerHTML = '';

      folderPath.forEach((item, index) => {

        const isLast = index === folderPath.length - 1;

       

        const btn = document.createElement('button');

        btn.className = `flex items-center gap-1 hover:text-indigo-400 transition-colors ${isLast ? 'font-semibold text-white' : 'text-gray-400'}`;

        btn.innerHTML = index === 0 ? `<i data-lucide="home" class="w-4 h-4"></i> ${item.name}` : item.name;

        btn.onclick =
